وأسوق مثلاً أخرو أخيرا، لاختاف الرأي الموهوم، بين تختلف العبرات في لفظها حتى إخاما امعنت النظرفي محلوا لإنها القينهانستوحف وحقا واحدا والمثل الحي أسوقله وهو اختلف القاطين بالفرحية والإجتماعية. ففي ظني أنه لوترك التقليل بين الطرفين وجخامطلاقان: القيوح، لإفرنته من معنا، فالمعنى البقيقي المقصود وهو الإنشط الفرح في ميادين العمل والفكر الإيماء عساله أن تحم المجموع لكنه خانفسلها ينفعى أن يحون الفرح في تشاطله فرحةا وإنما وهو مطلب بنوع معين من النشاط الخي صقق به فرحيته والذي يفيح المجتمع في الوقت نفسه إخ قح يتشاط الفرح بمايهحم المجتمع واخن فليس الشرط وهو الإنشط الفرح من حيث وهو فرح، بل الشرط هو أن يوجه نشاطله الفرحي غو نحمة الناس، افرض أن الفرح الذي نحاطبه بوهذا الحليم نسترف مهمةا للحكم في لعبة الحركة فحيف يمكن أن يمارس سرفته إلمن حيث وه فرحه لهخافتين للنطالبه بأن تعح من فرحيته بل نطالبله بيان وجه نشاطله الفرحي في احالة لمرفضه غووفــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــــ
